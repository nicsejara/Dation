def parse_failures(name):
    log=(EV/(name+'.log')).read_text()
    pattern=r'^(FAIL|ERROR): ([^\n]+)\n-{10,}\n(.*?)(?=^={20,}\n(?:FAIL|ERROR):|^-{20,}\nRan |\Z)'
    found=[]
    for kind,title,trace in re.findall(pattern,log,re.M|re.S):
        ident=re.search(r'\(([A-Za-z0-9_.]+)\)',title)
        ident=ident.group(1) if ident else title
        frames=re.findall(r'File "(/workspace/Dation/backend/[^\"]+)", line (\d+), in ([^\n]+)',trace)
        first=next((f for f in frames if '/tests/' in f[0]),None)
        assertion='';source=None
        if first:
            path=Path(first[0]);lineno=int(first[1]);source=str(path.relative_to(ROOT))+':'+str(lineno)
            assertion=path.read_text().splitlines()[lineno-1].strip()
        error_lines=[l for l in trace.splitlines() if re.match(r'^(AssertionError|KeyError|RuntimeError|TimeoutError|playwright\.).*',l)]
        observed=error_lines[-1] if error_lines else trace.strip().splitlines()[-1]
        category='Hallazgo nuevo en baseline; causa pendiente de aislamiento'
        classification='Hallazgo del baseline actual, origen aún sin atribuir. No hubo cambio productivo T0; no hay prueba histórica suficiente para exención G-B.'
        expected=assertion or 'La prueba debía completar el contrato indicado sin error.'
        recommendation='Reproducir el caso con inputs/selector/estado exactos; corregir sólo la causa demostrada dentro del alcance autorizado. No cambiar producto/asserts a ciegas.'
        blocking=True
        if "Input should be 'data_owner' or 'decision_owner'" in trace:
            category='Fixture/API de roles desactualizada'
            classification='Preexistente y demostrable por contrato actual: caller usa enum rechazado; hallazgo medido por primera vez en esta suite T0.'
            expected='Request de equipo aceptado por el test (200/201); fixture envía roles antiguos.'
            recommendation='Alinear sólo el fixture de escritura con data_owner/decision_owner y flujo actual, validando el escenario completo. No ampliar el enum público para hacer pasar el test.'
        elif "KeyError: 'data_steward'" in trace or "'data_steward\\ndata_owner' != 'data_steward'" in trace:
            category='Expectativa legacy de roles/proyección/migración'
            classification='Preexistente por diferencia explícita entre fixture y contrato/migraciones actuales; no producida por cambios T0.'
            recommendation='Conciliar aliases legacy de lectura, roles públicos actuales y bridge de migración; preservar persona/actor/versiones. No borrar roles o historia para satisfacer snapshots.'
        elif 'attributes_applied_at' in trace:
            category='Snapshot/fixture de Case anterior a atributos aplicados'
            classification='Preexistente: schema actual agrega el campo antes de T0; comparación del test no contempla ese estado.'
            recommendation='Actualizar el golden/fixture explícito de columnas del schema actual y verificar que los campos históricos permanecen iguales; no modificar filas productivas.'
        elif 'Aplicá los atributos del Case' in trace:
            category='Precondición/orden de gates anterior al contrato actual'
            classification='Preexistente demostrable: test espera un gate posterior sin satisfacer atributos actuales.'
            recommendation='Preparar atributos/Owner requeridos en la fixture y conservar las verificaciones posteriores; no relajar guards.'
        elif ident.endswith('test_assignment_scheduling_and_replacement_update_history_and_compact_list'):
            category='Conteo histórico previo a invalidación propia/downstream'
            classification='Preexistente: test espera4 eventos, observa5; SQL actual y escenario real A–F preservan supersession propia y downstream.'
            expected='Exactamente4 eventos de aprobación/supersession en el test.'
            observed='5 eventos. No se borró historia ni se cambió el resultado para cumplir el conteo.'
            recommendation='Reconciliar el golden de eventos con cada acción/actor/Run del flujo vigente; no sustituirlo por un conteo laxo.'
        elif ident.endswith('test_09_home_attention_tracks_real_versions_and_removes_approved_or_archived'):
            category='Expectativa de aprobación vigente anterior a nueva admission'
            classification='Preexistente: comentario del test retiene approval anterior tras admission; contrato actual retira propia/downstream. No es una decisión nueva T0.'
            recommendation='Reconciliar assertions con freshness/admission vigentes y source Run exacta; preservar historia y actores.'
        if "'Decision Owner'" in trace and "'Responsable'" in trace:
            category='Expectativa de labels anterior al vocabulario UI vigente'
            classification='Discrepancia preexistente de expectativa literal frente al componente actual; no se cambia la UI aprobada para hacer pasar el test.'
            recommendation='Reconciliar fixture/labels con Decision Owner y verificar accesibilidad/controles reales; no rediseñar ni relajar las verificaciones restantes.'
        if 'data-workflow-responsibility=data_steward' in trace:
            category='Selector legacy de responsabilidad'
            classification='Fixture/assertion refiere data_steward; contrato público actual usa data_owner. Confirmar el selector correcto antes de reparar.'
            recommendation='Actualizar únicamente fixture/selector demostrado obsoleto manteniendo el control de identidad, permisos y estado.'
        historical_signature=(ident.endswith('test_dashboard_trace_shares_the_copy_row_and_context_pills_are_equal') and '162.609375 not less than 1' in trace) or (ident.endswith('test_dda_navigation_resolves_canonical_runs_without_losing_context') and '[data-scheduling-review]' in trace and 'Timeout 6000ms' in trace) or (ident.endswith('test_map_assignment_is_consumed_readonly_by_config_and_review') and '[data-config-review]' in trace and 'Timeout 6000ms' in trace)
        if ident in historical and historical_signature:
            category='Deuda histórica WorkflowContext documentada'
            classification='ID registrado en summary versionado del baseline7351632; reproducido ahora en ef99723. No se generaliza esta exención a otros tests.'
            blocking=False
            recommendation='Reparación acotada al primer consumer afectado (T5 para routing/Config; T10 para geometría), con cobertura vigente; T11 exige suites verdes.'
        found.append({'suite':name,'kind':kind,'test_id':ident,'display_id':title,'component':ident.rsplit('.',1)[0],
                      'source':source,'assertion':assertion,'expected':expected,'observed':observed,
                      'category':category,'classification':classification,'impact':'Escenario de regresión sin certificar; revisar la expectativa y sus precondiciones antes de continuar.' if blocking else 'Deuda histórica específica con reparación/cobertura requerida por G-B.',
                      'historical_proof':'docs/final-polish-run-lineage-saga-summary.md:72; misma firma de fallo reproducida' if ident in historical and historical_signature else None,
                      'blocks_T1_individually':blocking,
                      'recommendation':recommendation,'evidence':[name+'.log','initial source-manifest.json',
                      'all839 tracked files equal prior audit manifest and current HEAD'],
                      'traceback':trace.strip()})
    return found

failures=[]
for r in results:
    if r['suite'] in ['B-BROWSER','B-E2E']:
        entries=parse_failures(r['suite']);failures.extend(entries)
        log=(EV/(r['suite']+'.log')).read_text()
        summary=re.search(r'^FAILED \(([^\n]+)\)',log,re.M)
        numbers={k:int(v) for k,v in re.findall(r'(failures|errors|skipped)=(\d+)',summary.group(1) if summary else '')}
        r.update(assertion_failures=numbers.get('failures',0),error_entries=numbers.get('errors',0),
                 failure_entries=len(entries),failed_test_ids_unique=len({x['test_id'] for x in entries}),
                 failure_display_ids=[x['display_id'] for x in entries])
        assert len(entries)==numbers.get('failures',0)+numbers.get('errors',0),(r['suite'],len(entries),numbers)
        write(EV/(r['suite']+'.result.json'),r)

identities={}
for r in results:
    log=(EV/(r['suite']+'.log')).read_text()
    if r['suite']=='B-NODE':
        ids=[arg for arg in r['command'] if arg.startswith('tests/')]
        assert all(('✔ '+arg+' ') in log for arg in ids)
    else:
        ids=re.findall(r'^(?:test\w+|unittest\.case\.FunctionTestCase) \((test[^)\n]+)\)',log,re.M)
    assert len(ids)==r['tests'],(r['suite'],len(ids),r['tests'])
    failed={f['test_id'] for f in failures if f['suite']==r['suite']}
    assert failed.issubset(set(ids)),(r['suite'],failed-set(ids))
    identities[r['suite']]=[{'test_id':i,'outcome':'FAIL' if i in failed else 'PASS',
                             'failure_entries':sum(f['suite']==r['suite'] and f['test_id']==i for f in failures)} for i in ids]
    r['failed_test_cases_unique']=len(failed)
    r['passed_test_cases']=len(ids)-len(failed)
    write(EV/(r['suite']+'.result.json'),r)
write(EV/'test-identities.json',identities)
